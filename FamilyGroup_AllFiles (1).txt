class Person {
  String id;
  String name, father, mother, photo, mobile, cnic, birthPlace, address,
      permAddress, gender, marital, spouse, relation, notes;
  DateTime? dob, dod;
  List<String> children;
  bool deceased;

  Person({
    required this.id,
    this.name = '',
    this.father = '',
    this.mother = '',
    this.photo = '',
    this.mobile = '',
    this.cnic = '',
    this.birthPlace = '',
    this.address = '',
    this.permAddress = '',
    this.gender = 'M',
    this.marital = 'ms_single',
    this.spouse = '',
    this.relation = 'other',
    this.notes = '',
    this.dob,
    this.dod,
    this.children = const [],
    this.deceased = false,
  });

  int? get age {
    if (dob == null) return null;
    final end = (deceased && dod != null) ? dod! : DateTime.now();
    var a = end.year - dob!.year;
    if (end.month < dob!.month ||
        (end.month == dob!.month && end.day < dob!.day)) {
      a--;
    }
    return a < 0 ? 0 : a;
  }

  Map<String, dynamic> toJson() => {
        'id': id,
        'name': name,
        'father': father,
        'mother': mother,
        'photo': photo,
        'mobile': mobile,
        'cnic': cnic,
        'birthPlace': birthPlace,
        'address': address,
        'permAddress': permAddress,
        'gender': gender,
        'marital': marital,
        'spouse': spouse,
        'relation': relation,
        'notes': notes,
        'dob': dob?.toIso8601String(),
        'dod': dod?.toIso8601String(),
        'children': children,
        'deceased': deceased,
      };

  factory Person.fromJson(Map<String, dynamic> j) => Person(
        id: j['id'] ?? '',
        name: j['name'] ?? '',
        father: j['father'] ?? '',
        mother: j['mother'] ?? '',
        photo: j['photo'] ?? '',
        mobile: j['mobile'] ?? '',
        cnic: j['cnic'] ?? '',
        birthPlace: j['birthPlace'] ?? '',
        address: j['address'] ?? '',
        permAddress: j['permAddress'] ?? '',
        gender: j['gender'] ?? 'M',
        marital: j['marital'] ?? 'ms_single',
        spouse: j['spouse'] ?? '',
        relation: j['relation'] ?? 'other',
        notes: j['notes'] ?? '',
        dob: j['dob'] != null ? DateTime.tryParse(j['dob']) : null,
        dod: j['dod'] != null ? DateTime.tryParse(j['dod']) : null,
        children: List<String>.from(j['children'] ?? const []),
        deceased: j['deceased'] ?? false,
      );
}

import 'dart:convert';
import 'dart:io';
import 'dart:math';
import 'package:crypto/crypto.dart';
import 'package:encrypt/encrypt.dart' as enc;

/// Encrypted local storage (AES-256) + password hashing.
class Vault {
  static const _s = FlutterSecureStorage();

  static String _hash(String pw, String salt) {
    var h = utf8.encode('$salt:$pw');
    for (var i = 0; i < 20000; i++) {
      h = sha256.convert(h).bytes;
    }
    return base64Encode(h);
  }

  static Future<bool> hasPassword() async =>
      (await _s.read(key: 'pw_hash')) != null;

  static Future<void> setPassword(String pw) async {
    final r = Random.secure();
    final salt = base64Encode(List<int>.generate(16, (_) => r.nextInt(256)));
    await _s.write(key: 'pw_salt', value: salt);
    await _s.write(key: 'pw_hash', value: _hash(pw, salt));
  }

  static Future<bool> verify(String pw) async {
    final salt = await _s.read(key: 'pw_salt');
    final hash = await _s.read(key: 'pw_hash');
    if (salt == null || hash == null) return false;
    return _hash(pw, salt) == hash;
  }

  static Future<enc.Key> _key() async {
    var k = await _s.read(key: 'data_key');
    if (k == null) {
      k = base64Encode(enc.Key.fromSecureRandom(32).bytes);
      await _s.write(key: 'data_key', value: k);
    }
    return enc.Key(base64Decode(k));
  }

  static Future<Directory> dir() => getApplicationDocumentsDirectory();

  static Future<File> _file() async =>
      File('${(await dir()).path}/family.dat');

  static Future<List<Person>> load() async {
    final f = await _file();
    if (!await f.exists()) return [];
    try {
      final parts = (await f.readAsString()).split(':');
      final e = enc.Encrypter(enc.AES(await _key(), mode: enc.AESMode.cbc));
      final txt = e.decrypt64(parts[1], iv: enc.IV.fromBase64(parts[0]));
      return (jsonDecode(txt) as List)
          .map((x) => Person.fromJson(Map<String, dynamic>.from(x)))
          .toList();
    } catch (_) {
      return [];
    }
  }

  static Future<void> save(List<Person> people) async {
    final iv = enc.IV.fromSecureRandom(16);
    final e = enc.Encrypter(enc.AES(await _key(), mode: enc.AESMode.cbc));
    final data = e.encrypt(jsonEncode(people.map((p) => p.toJson()).toList()),
        iv: iv);
    await (await _file()).writeAsString('${iv.base64}:${data.base64}');
  }
}

import 'package:flutter/material.dart';
import 'package:shared_preferences/shared_preferences.dart';
import '../models/person.dart';
import '../services/vault.dart';
import 'strings.dart';

class AppState extends ChangeNotifier {
  List<Person> people = [];
  ThemeMode themeMode = ThemeMode.system;
  String lang = 'ur';
  bool loggedIn = false;
  bool hasPassword = false;
  late SharedPreferences _p;

  String tr(String k) => trLang(lang, k);

  Future<void> init() async {
    _p = await SharedPreferences.getInstance();
    lang = _p.getString('lang') ?? 'ur';
    themeMode = ThemeMode.values[_p.getInt('theme') ?? 0];
    hasPassword = await Vault.hasPassword();
  }

  Future<bool> login(String pw) async {
    if (!await Vault.verify(pw)) return false;
    people = await Vault.load();
    loggedIn = true;
    notifyListeners();
    return true;
  }

  Future<void> setupPassword(String pw) async {
    await Vault.setPassword(pw);
    hasPassword = true;
    people = await Vault.load();
    loggedIn = true;
    notifyListeners();
  }

  Future<bool> changePassword(String old, String nw) async {
    if (!await Vault.verify(old)) return false;
    await Vault.setPassword(nw);
    return true;
  }

  void logout() {
    loggedIn = false;
    people = [];
    notifyListeners();
  }

  Future<void> upsert(Person p) async {
    final i = people.indexWhere((x) => x.id == p.id);
    if (i >= 0) {
      people[i] = p;
    } else {
      people.add(p);
    }
    await Vault.save(people);
    notifyListeners();
  }

  Future<void> remove(String id) async {
    people.removeWhere((x) => x.id == id);
    await Vault.save(people);
    notifyListeners();
  }

  void setLang(String l) {
    lang = l;
    _p.setString('lang', l);
    notifyListeners();
  }

  void setTheme(ThemeMode m) {
    themeMode = m;
    _p.setInt('theme', m.index);
    notifyListeners();
  }
}

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});
  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final _p = TextEditingController();
  final _c = TextEditingController();
  String? _err;

  Future<void> _submit() async {
    final s = context.read<AppState>();
    if (!s.hasPassword) {
      if (_p.text.length < 6) return setState(() => _err = s.tr('pw_short'));
      if (_p.text != _c.text) return setState(() => _err = s.tr('pw_mismatch'));
      await s.setupPassword(_p.text);
    } else {
      if (!await s.login(_p.text)) setState(() => _err = s.tr('pw_wrong'));
    }
  }

  @override
  Widget build(BuildContext context) {
    final s = context.watch<AppState>();
    final setup = !s.hasPassword;
    final cs = Theme.of(context).colorScheme;
    return Directionality(
      textDirection: s.lang == 'ur' ? TextDirection.rtl : TextDirection.ltr,
      child: Scaffold(
        body: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: Column(children: [
                Icon(Icons.family_restroom, size: 88, color: cs.primary),
                const SizedBox(height: 8),
                Text(s.tr('app'),
                    style: Theme.of(context).textTheme.headlineMedium),
                const SizedBox(height: 24),
                if (setup)
                  Padding(
                      padding: const EdgeInsets.only(bottom: 12),
                      child: Text(s.tr('setup_title'),
                          style: Theme.of(context).textTheme.titleMedium)),
                TextField(
                    controller: _p,
                    obscureText: true,
                    decoration: InputDecoration(
                        labelText: s.tr('password'),
                        prefixIcon: const Icon(Icons.lock))),
                if (setup) ...[
                  const SizedBox(height: 12),
                  TextField(
                      controller: _c,
                      obscureText: true,
                      decoration: InputDecoration(
                          labelText: s.tr('confirm'),
                          prefixIcon: const Icon(Icons.lock_outline))),
                ],
                if (_err != null)
                  Padding(
                      padding: const EdgeInsets.only(top: 8),
                      child: Text(_err!, style: TextStyle(color: cs.error))),
                const SizedBox(height: 20),
                SizedBox(
                    width: double.infinity,

                        onPressed: _submit,

int daysToBirthday(DateTime dob) {
  final n = DateTime.now();
  final today = DateTime(n.year, n.month, n.day);
  DateTime occ(int y) =>
      DateTime(y, dob.month, (dob.month == 2 && dob.day == 29) ? 28 : dob.day);
  var next = occ(n.year);
  if (next.isBefore(today)) next = occ(n.year + 1);
  return next.difference(today).inDays;
}

class PersonForm extends StatefulWidget {
  final Person? existing;
  const PersonForm({super.key, this.existing});
  @override
  State<PersonForm> createState() => _PersonFormState();
}

class _PersonFormState extends State<PersonForm> {
  final _key = GlobalKey<FormState>();
  late final Map<String, TextEditingController> _c;
  late String _gender, _marital, _relation, _photo;
  DateTime? _dob, _dod;
  late bool _deceased;
  static const _fields = [
    'name', 'father', 'mother', 'mobile', 'cnic', 'birthPlace', 'address',
    'permAddress', 'spouse', 'children', 'notes'
  ];

  @override
  void initState() {
    super.initState();
    final p = widget.existing;
    final init = <String, String>{
      'name': p?.name ?? '',
      'father': p?.father ?? '',
      'mother': p?.mother ?? '',
      'mobile': p?.mobile ?? '',
      'cnic': p?.cnic ?? '',
      'birthPlace': p?.birthPlace ?? '',
      'address': p?.address ?? '',
      'permAddress': p?.permAddress ?? '',
      'spouse': p?.spouse ?? '',
      'children': (p?.children ?? []).join(', '),
      'notes': p?.notes ?? '',
    };
    _c = {for (final f in _fields) f: TextEditingController(text: init[f])};
    _gender = p?.gender ?? 'M';
    _marital = p?.marital ?? 'ms_single';
    _relation = p?.relation ?? 'other';
    _photo = p?.photo ?? '';
    _dob = p?.dob;
    _dod = p?.dod;
    _deceased = p?.deceased ?? false;
  }

  Future<void> _pickPhoto() async {
    final x = await ImagePicker()
        .pickImage(source: ImageSource.gallery, maxWidth: 800, imageQuality: 80);
    if (x == null) return;
    final d = Directory('${(await Vault.dir()).path}/photos');
    await d.create(recursive: true);
    final path = '${d.path}/${DateTime.now().millisecondsSinceEpoch}.jpg';
    await File(x.path).copy(path);
    setState(() => _photo = path);
  }

  Future<void> _pickDate(bool birth) async {
    final d = await showDatePicker(
      context: context,
      initialDate: (birth ? _dob : _dod) ?? DateTime(1990),
      firstDate: DateTime(1900),
      lastDate: DateTime.now(),
    );
    if (d != null) setState(() => birth ? _dob = d : _dod = d);
  }

  Future<void> _save() async {
    if (!_key.currentState!.validate()) return;
    final s = context.read<AppState>();
    final nav = Navigator.of(context);
    String v(String k) => _c[k]!.text.trim();
    final p = Person(
      id: widget.existing?.id ?? DateTime.now().microsecondsSinceEpoch.toString(),
      name: v('name'),
      father: v('father'),
      mother: v('mother'),
      photo: _photo,
      mobile: v('mobile'),
      cnic: v('cnic'),
      birthPlace: v('birthPlace'),
      address: v('address'),
      permAddress: v('permAddress'),
      gender: _gender,
      marital: _marital,
      spouse: v('spouse'),
      relation: _relation,
      notes: v('notes'),
      dob: _dob,
      dod: _deceased ? _dod : null,
      deceased: _deceased,
      children: v('children')
          .split(RegExp(r'[,،\n]'))
          .map((e) => e.trim())
          .where((e) => e.isNotEmpty)
          .toList(),
    );
    await s.upsert(p);
    nav.pop();
  }

  Widget _tf(AppState s, String k, String label,
      {TextInputType? type, bool req = false, int lines = 1}) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 12),
      child: TextFormField(
        controller: _c[k],
        keyboardType: type,
        maxLines: lines,
        decoration: InputDecoration(labelText: s.tr(label)),
        validator: req
            ? (x) => (x == null || x.trim().isEmpty) ? s.tr('required') : null
            : null,
      ),
    );
  }

  Widget _dateTile(AppState s, String label, DateTime? d, bool birth) =>
      Padding(
        padding: const EdgeInsets.only(bottom: 12),
        child: OutlinedButton.icon(
          icon: const Icon(Icons.calendar_today),
          label: Text('${s.tr(label)}: ${d == null ? s.tr('pick') : fmtDate(d)}'),
          onPressed: () => _pickDate(birth),
        ),
      );

  @override
  Widget build(BuildContext context) {
    final s = context.watch<AppState>();
    final preview = Person(id: 'x', name: _c['name']!.text, photo: _photo);
    return Directionality(
      textDirection: s.lang == 'ur' ? TextDirection.rtl : TextDirection.ltr,
      child: Scaffold(
        appBar: AppBar(
          title: Text(s.tr(widget.existing == null ? 'add_person' : 'edit')),
          actions: [
            TextButton(onPressed: _save, child: Text(s.tr('save'))),
          ],
        ),
        body: Form(
          key: _key,
          child: ListView(padding: const EdgeInsets.all(16), children: [
            Center(
                child: GestureDetector(
                    onTap: _pickPhoto,
                    child: PersonAvatar(preview, radius: 52))),
            TextButton.icon(
                onPressed: _pickPhoto,
                icon: const Icon(Icons.photo),
                label: Text(s.tr('photo'))),
            const SizedBox(height: 8),
            _tf(s, 'name', 'name', req: true),
            DropdownButtonFormField<String>(
              initialValue: _relation,
              decoration: InputDecoration(labelText: s.tr('relation')),
              items: [
                for (final r in relations)
                  DropdownMenuItem(value: r, child: Text(s.tr('rel_$r')))
              ],
              onChanged: (v) => setState(() => _relation = v!),
            ),
            const SizedBox(height: 12),
            SegmentedButton<String>(
              segments: [
                ButtonSegment(value: 'M', label: Text(s.tr('male'))),
                ButtonSegment(value: 'F', label: Text(s.tr('female'))),
              ],
              selected: {_gender},
              onSelectionChanged: (v) => setState(() => _gender = v.first),
            ),
            const SizedBox(height: 12),
            _tf(s, 'father', 'father'),
            _tf(s, 'mother', 'mother'),
            _tf(s, 'mobile', 'mobile', type: TextInputType.phone),

## APK اور AAB بنانے کا طریقہ
1. GitHub پر نئی repository بنائیں اور اس پوری فولڈر کو اپلوڈ کریں۔
2. Actions ٹیب → "Build Android APK and AAB" → Run workflow۔
3. مکمل ہونے پر Artifacts سے `family-group-release` ڈاؤن لوڈ کریں (APK + AAB)۔

یا اپنے کمپیوٹر پر: `flutter create --platforms=android .` پھر `flutter run`۔

## Play Store
AAB پر اپلوڈ کی (upload keystore) سے سائن کرنا ہوگا۔ keystore بنائیں:
`keytool -genkey -v -keystore upload.jks -keyalg RSA -keysize 2048 -validity 10000 -alias upload`
پھر `android/key.properties` اور `android/app/build.gradle` میں signingConfig شامل کریں
(Flutter ڈاکس: docs.flutter.dev/deployment/android)۔ اس کے بغیر بلڈ debug key سے سائن ہوتا ہے۔
